import { test, expect } from '@playwright/test';
import fs from 'node:fs';
import path from 'node:path';
import { bootstrapUnlockedSession, openPatientSection } from '../e2e/utils';
import { seedDemoData, post } from './seed';
const out = process.env.REAL_COMMAND_OUT!;
const phase = process.env.REAL_COMMAND_PHASE!;
const rows: unknown[] = [];
test('viste reali dei comandi, confronto sintetico', async ({ page }) => {
  test.setTimeout(240_000);
  await bootstrapUnlockedSession(page, '1234');
  await seedDemoData(page);
  await post(page, '/api/therapies', { id: 'audit-suspended', patientId: 'readme-shots-giulia-fervi', drugName: 'Amlodipina sintetica', dosage: '5 mg la sera', status: 'suspended', startDate: '2025-01-01T00:00:00Z' });
  await post(page, '/api/entries', { id: 'audit-allergy', patientId: 'readme-shots-giulia-fervi', type: 'note', title: 'Allergia sintetica documentata', content: 'Allergia ad amoxicillina: orticaria. Caso inventato per audit visivo.', date: '2026-10-21T09:00:00Z' });
  await post(page, '/api/entries', { id: 'audit-scale', patientId: 'readme-shots-giulia-fervi', type: 'scale', title: 'ADL (Katz) — valutazione sintetica', content: 'ADL 5/6. Assistenza per il bagno. Valutazione inventata.', date: '2026-10-21T09:00:00Z', metadata: { scaleId: 'adl', score: 5, totalScore: 5, maxScore: 6 } });
  const visit = async (url: string) => { await page.goto(url); await page.waitForLoadState('networkidle'); };
  const shot = async (state: string, heights: number[]) => {
    for (const [i, width] of [1440, 768].entries()) for (const theme of ['giorno', 'grafite']) {
      await page.setViewportSize({ width, height: heights[i] });
      await page.emulateMedia({ colorScheme: theme === 'giorno' ? 'light' : 'dark', reducedMotion: 'reduce' });
      await page.evaluate(theme => {
        const mode = theme === 'giorno' ? 'light' : 'dark';
        localStorage.setItem('mediflow-theme', mode);
        document.documentElement.classList.remove('light', 'dark');
        document.documentElement.classList.add(mode);
      }, theme);
      await page.evaluate(() => document.fonts.ready);
      await page.evaluate(() => { document.querySelectorAll<HTMLElement>('*').forEach(e => { if(e.scrollTop) e.scrollTop = 0; }); window.scrollTo(0, 0); });
      await page.mouse.move(0, 0);
      await page.waitForTimeout(250);
      const metrics = await page.evaluate(() => [...document.querySelectorAll<HTMLElement>('.ui-btn-primary,.mf-btn-secondary,.ui-btn-secondary,.mf-button-primary,.mf-button-secondary,.mf-button-text,.mf-button-icon,.mf-button-destructive')].filter(e => e.getBoundingClientRect().width && e.getBoundingClientRect().height).map(e => {
        const r=e.getBoundingClientRect(),s=getComputedStyle(e);
        return { text: e.innerText || e.getAttribute('aria-label'), x:r.x,y:r.y,width:r.width,height:r.height,padding:s.padding,radius:s.borderRadius,position:s.position,overflow:s.overflow,whiteSpace:s.whiteSpace,
          absoluteChildren:[...e.querySelectorAll<HTMLElement>('*')].filter(c=>getComputedStyle(c).position==='absolute').map(c=>({class:c.getAttribute('class'),text:c.innerText})) };
      }));
      const file = `real-${phase}-${state}-${theme}-${width}.png`;
      fs.mkdirSync(out, {recursive:true});
      await page.screenshot({path:path.join(out,file),fullPage:true,animations:'disabled'});
      rows.push({state,theme,width,height:heights[i],file,url:new URL(page.url()).pathname+new URL(page.url()).hash,metrics});
      fs.writeFileSync(path.join(out,`real-${phase}-metrics.json`),JSON.stringify(rows,null,2)+'\n');
      console.log('CAPTURE',file);
    }
  };
  const full='/patients/readme-shots-giulia-fervi';
  await visit(full+'/modules');
  await expect(page.getByRole('heading',{name:/Giulia/}).first()).toBeVisible();
  await shot('011-riepilogo',[1425,2241]);
  await openPatientSection(page,'terapie');
  await page.locator('#terapie').getByRole('button',{name:'Nuova terapia',exact:true}).click();
  await expect(page.getByRole('heading',{name:'Nuova terapia',exact:true})).toBeVisible();
  await shot('054-nuova-terapia',[2060,2230]);
  await visit('/settings');
  await expect(page.getByRole('heading',{name:'Impostazioni',exact:true})).toBeVisible();
  await shot('impostazioni-indice',[1100,1600]);
  await visit(full+'/modules');
  await openPatientSection(page,'terapie');
  await page.getByRole('button',{name:'Elimina Ramipril solo se inserito per errore',exact:true}).click();
  await expect(page.getByRole('dialog')).toBeVisible();
  await shot('conferma-elimina-terapia',[960,960]);
  await page.getByRole('dialog').getByRole('button',{name:/Annulla|Torna/}).first().click();
  await post(page,'/api/auth/reset',{});
  await visit('/');
  await expect(page.getByLabel('Nome e cognome',{exact:true})).toBeVisible();
  await page.getByLabel('Nome e cognome',{exact:true}).fill('Dr. Demo Sintetico');
  await page.getByLabel('Nome dell’ambulatorio',{exact:true}).fill('Ambulatorio Audit Sintetico');
  await page.getByRole('button',{name:'Avanti',exact:true}).click();
  await expect(page.getByLabel('Passphrase di accesso',{exact:true})).toBeVisible();
  await shot('onboarding-proteggi-accesso',[960,960]);
});
