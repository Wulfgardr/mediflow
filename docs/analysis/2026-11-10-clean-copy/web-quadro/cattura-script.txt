/* Catture prima/dopo del Quadro paziente Web (clean-copy). Solo dati inventati,
   trasporti finti, nessun modello vero. Lo script è lo stesso su origin/main e sul ramo. */
import fs from 'node:fs';
import path from 'node:path';
import { expect, test, type Locator, type Page } from '@playwright/test';

import type { FunctionModelPreferences } from '../lib/function-models/browser';
import { parsePatientInsightPreviewRequest, type PatientInsightPreviewRequest } from '../lib/ai-providers/fabric/patient-insight-preview-contract';
import { installOrdinaryChatGptTransport, remoteMetadata } from './ordinary-chatgpt-ui-fixture';
import { bootstrapUnlockedSession, openPatientSection, setAiLaneKillSwitch } from './utils';

const OUT = process.env.CLEAN_COPY_CAPTURE_DIR ?? '';
const MODEL = 'modello-sintetico:latest';
const OPTION = `model_option_${'a'.repeat(32)}`;
const DRUG = 'Farmaco sintetico cattura';
const CATALOG = {
  schemaVersion: 'mediflow.function-preferences.v1', revision: `sha256_${'c'.repeat(64)}`, catalogRevision: `sha256_${'d'.repeat(64)}`,
  check: 'configuration_only', apply: 'denied', presets: ['host_defaults', 'all_off'],
  functions: (['patient_insight', 'smart_import', 'document_synthesis', 'treatment_reasoning'] as const).map(id => ({
    id, enabled: id === 'patient_insight', defaultModelOptionId: OPTION, defaultSource: 'host_configuration', bindingState: 'current',
    options: [{ modelOptionId: OPTION, label: MODEL, provider: 'ollama', state: 'available_unqualified' }],
  })),
} satisfies FunctionModelPreferences;

function proposal(request: { capturedAt: string; patientRevision: number }) {
  return {
    writesPerformed: 0, apply: 'denied', status: 'available', code: null,
    proposal: {
      schemaVersion: 'mediflow.patient-insight.review-proposal.v2', reviewOnly: true,
      summary: 'Sintesi inventata da rivedere. [S1]', currentState: ['Quadro inventato. [S1]'], alerts: ['Attenzione inventata. [S1]'], nextSteps: ['Controllo inventato. [S1]'], gaps: [],
      generatedAt: request.capturedAt,
      currentness: { selectionEpoch: 3, patientRevision: request.patientRevision, projectionDigest: `sha256_${'a'.repeat(64)}`, capturedAt: request.capturedAt, verifiedAt: request.capturedAt },
    },
  };
}

async function shot(target: Locator, width: number, name: string) {
  if (!OUT) return;
  fs.mkdirSync(OUT, { recursive: true });
  await target.screenshot({ path: path.join(OUT, `${name}-${width}.png`), animations: 'disabled' });
}

async function newPatient(page: Page, withTherapy = false) {
  return page.evaluate(async ({ drug, withTherapy }) => {
    const post = async (p: string, body: unknown) => {
      const r = await fetch(`/api/${p}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) });
      if (!r.ok) throw new Error(`${p}: ${r.status}`);
      return r.json() as Promise<{ id: string }>;
    };
    const patient = await post('patients', { firstName: 'Cattura', lastName: 'Sintetica', taxCode: `CCAT${crypto.randomUUID().replaceAll('-', '').slice(0, 12).toUpperCase()}`, birthDate: '1970-05-04T00:00:00.000Z' });
    if (withTherapy) await post('therapies', { id: `therapy-${crypto.randomUUID()}`, patientId: patient.id, drugName: drug, activePrinciple: 'Principio inventato', motivation: 'Nota inventata', dosage: 'Dose inventata', status: 'active', startDate: '2026-10-01T08:00:00.000Z' });
    return patient.id;
  }, { drug: DRUG, withTherapy });
}

async function openInsight(page: Page, patientId: string) {
  await page.goto(`/patients/${patientId}/modules`);
  await openPatientSection(page, 'attenzione');
  const panel = page.locator('#patient-insight');
  await panel.locator(':scope > div > [data-disclosure-control]').click();
  return panel;
}

test.describe.configure({ retries: 0 });

for (const width of [1440, 390]) {
  test(`cattura ${width}: funzione spenta`, async ({ page }) => {
    await page.setViewportSize({ width, height: 900 });
    await bootstrapUnlockedSession(page, process.env.E2E_PIN || '1234');
    await setAiLaneKillSwitch(page, 'aiPatientInsightKillSwitch', 'disabled');
    const panel = await openInsight(page, await newPatient(page));
    await expect(panel.getByTestId('patient-insight-disabled-card')).toBeVisible();
    await shot(panel, width, '01-funzione-spenta');
    await setAiLaneKillSwitch(page, 'aiPatientInsightKillSwitch', 'enabled');
  });

  test(`cattura ${width}: locale, pronto, selettore, proposta, superata`, async ({ page }) => {
    await page.setViewportSize({ width, height: 900 });
    await bootstrapUnlockedSession(page, process.env.E2E_PIN || '1234');
    await setAiLaneKillSwitch(page, 'aiPatientInsightKillSwitch', 'enabled');
    const patientId = await newPatient(page, true);
    await page.route('**/api/settings/ai/functions', route => route.fulfill({ status: 200, json: CATALOG }));
    await page.route('**/api/ai/patient-insight/preview', async route => {
      const request = parsePatientInsightPreviewRequest(route.request().postDataJSON()) as PatientInsightPreviewRequest;
      const { receipt, provenance } = { receipt: { schemaVersion: 'mediflow.ai.fabric-resolution.v1', capability: 'patient_insight', venue: 'local_process', provider: 'ollama', model: MODEL, egress: 'none' },
        provenance: { schemaVersion: 'mediflow.ai.fabric-provenance.v1', capability: 'patient_insight', venue: 'local_process', provider: 'ollama', model: MODEL, preprocessing: ['context_minimization', 'envelope_validation'] } };
      await route.fulfill({ status: 200, json: { preview: { ...proposal(request), receipt, provenance, reviewRef: `review_${'b'.repeat(32)}` } } });
    });
    const panel = await openInsight(page, patientId);
    await expect(panel.getByRole('button', { name: 'Cambia', exact: true })).toBeEnabled();
    await shot(panel, width, '02-pronto-a-generare');
    await panel.getByRole('button', { name: 'Cambia', exact: true }).click();
    const model = panel.getByRole('combobox', { name: 'Modello per questa proposta', exact: true });
    await model.focus(); await model.selectOption(OPTION);
    await panel.getByText('Dettagli', { exact: true }).first().click();
    await shot(panel, width, '03-selettore-locale-aperto');
    await panel.getByRole('button', { name: /^(Avvia supporto|Genera proposta)$/u }).click();
    await expect(panel.getByTestId('patient-insight-review-proposal')).toBeVisible();
    await shot(panel, width, '04-proposta');
    await openPatientSection(page, 'terapie');
    const row = page.locator('#terapie').getByRole('row').filter({ has: page.getByText(DRUG, { exact: true }) });
    await row.getByRole('button', { name: 'Modifica', exact: true }).click();
    await page.locator('#terapie input[name="dosage"]').fill('Dose cambiata inventata');
    const saved = page.waitForResponse(r => r.request().method() === 'PUT' && /\/api\/therapies\//u.test(r.url()));
    await page.locator('#terapie').getByRole('button', { name: 'Aggiorna terapia', exact: true }).click();
    await saved;
    await openPatientSection(page, 'attenzione');
    await expect(panel.getByTestId('patient-insight-superseded')).toBeVisible();
    await shot(panel, width, '05-proposta-superata');
  });

  test(`cattura ${width}: selettore OpenAI senza e con consenso`, async ({ page }) => {
    await page.setViewportSize({ width, height: 900 });
    let request: { capturedAt: string; patientRevision: number } | null = null;
    const remote = await installOrdinaryChatGptTransport(page, 'patient_insight', () => {
      const { receipt, provenance } = remoteMetadata('patient_insight');
      return { preview: { ...proposal(request!), receipt, provenance, reviewRef: `review_${'e'.repeat(32)}` } };
    });
    await bootstrapUnlockedSession(page, process.env.E2E_PIN || '1234');
    await setAiLaneKillSwitch(page, 'aiPatientInsightKillSwitch', 'enabled');
    const panel = await openInsight(page, await newPatient(page));
    await panel.getByRole('button', { name: 'Cambia', exact: true }).click();
    await panel.getByRole('combobox', { name: 'Modello per questa proposta' }).selectOption('chatgpt_subscription');
    const controls = panel.getByRole('group', { name: 'OpenAI per questa proposta' });
    const consent = controls.getByRole('checkbox', { name: /Consenti OpenAI/u });
    await expect(consent).not.toBeChecked();
    await shot(panel, width, '06-openai-senza-consenso');
    await consent.click(); await expect(consent).toBeChecked();
    await shot(panel, width, '07-openai-con-consenso');
    await panel.getByRole('button', { name: /^(Avvia supporto|Genera proposta)$/u }).click();
    await expect(controls).toContainText('Contesto preparato e redatto: 100 byte.');
    request = remote.previews[0] as typeof request;
    await controls.getByRole('button', { name: 'Consenti per questo contesto' }).click();
    await controls.getByRole('button', { name: 'Avvia accesso dedicato ChatGPT' }).click();
    await controls.getByRole('button', { name: 'Ho completato: verifica accesso' }).click();
    await shot(panel, width, '08-openai-accesso-fatto');
    await controls.getByRole('button', { name: /^(Leggi catalogo di questo processo|Leggi i modelli disponibili)$/u }).click();
    await expect(controls.getByRole('combobox', { name: 'Modello e ragionamento correnti' })).toBeVisible();
    await shot(panel, width, '09-openai-catalogo');
    await controls.getByRole('combobox', { name: 'Modello e ragionamento correnti' }).selectOption(remote.option());
    await controls.getByRole('button', { name: 'Genera questa proposta' }).click();
    await expect(page.getByTestId('patient-insight-review-proposal')).toBeVisible();
    await shot(panel, width, '10-openai-completato');
  });
}
